```bash
#!/usr/bin/env bash
# Confere que todo arquivo versionado nos repos antigos existe no DataCore, no
# lugar novo. Removidos de propósito (compose e traefik antigos) e os Dockerfiles
# renomeados estão na lista de exceções. Sai 1 e lista o que faltou.
set -euo pipefail
RAIZ="$HOME/github/DataCore"
FRONT_SHA="${FRONT_SHA:-a19b1e2f}"
BACK_SHA="${BACK_SHA:-b18e726}"
faltou=0

confere() {  # $1 = caminho esperado no DataCore
    if [ ! -e "$RAIZ/$1" ]; then
        echo "FALTA: $1"
        faltou=1
    fi
}

while IFS= read -r f; do
    case "$f" in
        docker-compose.yml|traefik/traefik.yml) continue ;;
        dockerfile) confere "frontend/Dockerfile" ;;
        *) confere "frontend/$f" ;;
    esac
done < <(git -C ~/github/DataCoreHS ls-tree -r --name-only "$FRONT_SHA")

while IFS= read -r f; do
    case "$f" in
        docker-compose.yml|traefik/*) continue ;;
        dockerfile) confere "backend/Dockerfile" ;;
        analytics/*|deploy/*) confere "$f" ;;
        *) confere "backend/$f" ;;
    esac
done < <(git -C ~/github/tiny-integrador ls-tree -r --name-only "$BACK_SHA")

if [ "$faltou" -eq 0 ]; then
    echo "OK: snapshot completo"
fi
exit "$faltou"
```

- [ ] **Step 2: `scripts/varrer.sh`**

```bash
#!/usr/bin/env bash
# Varre o que o git vai versionar (tracked + untracked não ignorado) atrás de
# credencial, IP e dado pessoal. Uso: scripts/varrer.sh [arquivo-de-valores]
# Sai 1 se achar algo. Os valores sensíveis vêm de fora do repo e nunca são impressos.
set -uo pipefail
RAIZ="$(cd "$(dirname "$0")/.." && pwd)"
VALORES="${1:-$HOME/.config/datacore-varredura.txt}"
cd "$RAIZ"
mapfile -t ARQS < <(git ls-files -co --exclude-standard)
achou=0

if [ -f "$VALORES" ]; then
    if grep -l -I -F -f "$VALORES" -- "${ARQS[@]}" 2>/dev/null; then
        echo "^ VALOR SENSÍVEL LITERAL nos arquivos acima"; achou=1
    fi
fi

padroes='postgres(ql)?://[^[:space:]:/]+:[^[:space:]@]+@|BEGIN [A-Z ]*PRIVATE KEY|(sk-[A-Za-z0-9]{20,}|ghp_[A-Za-z0-9]{20,}|re_[A-Za-z0-9]{20,})|[0-9]{3}\.[0-9]{3}\.[0-9]{3}-[0-9]{2}'
if grep -n -I -E "$padroes" -- "${ARQS[@]}" 2>/dev/null; then
    echo "^ PADRÃO DE CREDENCIAL/CPF — revisar cada linha"; achou=1
fi

proibidos=$(printf '%s\n' "${ARQS[@]}" | grep -E '(^|/)\.env$|(^|/)\.env\.[^e]|\.(xlsx|jsonl|pem|key|pfx|dump)$|acme\.json$|SETUP-CLAUDE\.md$' || true)
if [ -n "$proibidos" ]; then
    echo "$proibidos"; echo "^ ARQUIVO QUE NÃO DEVIA SER VERSIONADO"; achou=1
fi

[ "$achou" -eq 0 ] && echo "OK: nada encontrado"
exit "$achou"
```
